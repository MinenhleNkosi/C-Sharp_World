using System;

public class Solution
{
    public static void PrintEvenOrOdd(int number)
    {
        // Your code here
        // Print "Even" if the number is even
        // Print "Odd" if the number is odd
        if (number % 2 == 0)
        {
            Console.WriteLine("Even");
        }
        else
        {
            Console.WriteLine("Odd");
        }
    }
}
